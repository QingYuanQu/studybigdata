代码8-3  启动Hive远程服务
cd /usr/local/hive-3.1.2/bin
hive --service metastore &  //启动Hive元数据服务
jps  //查看后台服务进程
nohup hive --service hiveserver2 &  //后台启动HiveServer2服务
jps  //查看后台服务进程
